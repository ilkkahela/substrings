# 3.28 is Ubuntu 24.04's CMake; the range keeps newer policies on newer CMake.
cmake_minimum_required(VERSION 3.28...4.2)
project(substrings LANGUAGES C CXX)

# libsais builds the suffix array in parallel with OpenMP when it's available.
# GCC ships it; Clang needs its libomp package (see README).
option(SUBSTRINGS_USE_OPENMP "Build libsais with OpenMP if the compiler supports it" ON)
set(use_openmp OFF)
if(SUBSTRINGS_USE_OPENMP)
    find_package(OpenMP COMPONENTS C)
    if(OpenMP_C_FOUND)
        set(use_openmp ON)
    else()
        message(WARNING "OpenMP not found: suffix arrays will be built single-threaded")
    endif()
endif()
set(LIBSAIS_USE_OPENMP ${use_openmp} CACHE BOOL "" FORCE)

include(FetchContent)
FetchContent_Declare(
    libsais
    GIT_REPOSITORY https://github.com/IlyaGrebnov/libsais.git
    GIT_TAG v2.10.4
    GIT_SHALLOW TRUE
    EXCLUDE_FROM_ALL  # build it for substrings, but don't install its library and headers
)
FetchContent_MakeAvailable(libsais)

# Warnings for our own targets only, not for libsais.
set(SUBSTRINGS_WARNINGS -Wall -Wextra -Wpedantic)

add_library(substrings_core STATIC
    cli.cpp
    completions.cpp
    lz77.cpp
    palindromes.cpp
    suffix_array.cpp
    suffix_tree.cpp
    table_viz.cpp
    terminal.cpp
)
target_compile_features(substrings_core PUBLIC cxx_std_23)
target_link_libraries(substrings_core PUBLIC libsais)
target_compile_options(substrings_core PRIVATE ${SUBSTRINGS_WARNINGS})

add_executable(substrings main.cpp)
target_link_libraries(substrings PRIVATE substrings_core)
target_compile_options(substrings PRIVATE ${SUBSTRINGS_WARNINGS})

# Shell completions are generated by the built substrings from its option table.
set(completion_dir ${CMAKE_CURRENT_BINARY_DIR}/completions)
set(completion_files
    ${completion_dir}/substrings.bash ${completion_dir}/_substrings ${completion_dir}/substrings.fish)
add_custom_command(
    OUTPUT ${completion_files}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${completion_dir}
    COMMAND substrings --completions=bash -o ${completion_dir}/substrings.bash
    COMMAND substrings --completions=zsh -o ${completion_dir}/_substrings
    COMMAND substrings --completions=fish -o ${completion_dir}/substrings.fish
    DEPENDS substrings
    VERBATIM
)
add_custom_target(completions ALL DEPENDS ${completion_files})

include(GNUInstallDirs)
install(TARGETS substrings)
install(FILES substrings.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES ${completion_dir}/substrings.bash
        DESTINATION ${CMAKE_INSTALL_DATADIR}/bash-completion/completions RENAME substrings)
install(FILES ${completion_dir}/_substrings DESTINATION ${CMAKE_INSTALL_DATADIR}/zsh/site-functions)
install(FILES ${completion_dir}/substrings.fish
        DESTINATION ${CMAKE_INSTALL_DATADIR}/fish/vendor_completions.d)

find_package(GTest)
if(GTest_FOUND)
    enable_testing()
    add_executable(substrings_tests tests.cpp)
    target_link_libraries(substrings_tests PRIVATE substrings_core GTest::gtest_main)
    target_compile_options(substrings_tests PRIVATE ${SUBSTRINGS_WARNINGS})
    # The man page test reads substrings.1 from the source tree.
    target_compile_definitions(substrings_tests PRIVATE SUBSTRINGS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME substrings_tests COMMAND substrings_tests)
endif()
